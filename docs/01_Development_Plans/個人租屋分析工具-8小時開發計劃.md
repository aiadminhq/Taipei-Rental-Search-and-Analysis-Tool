```javascript
// mcp/client.js - MCP 客戶端
const { Client } = require('@modelcontextprotocol/sdk/client/index.js');
const { StdioClientTransport } = require('@modelcontextprotocol/sdk/client/stdio.js');

class MCPClient {
  constructor(config) {
    this.config = config;
    this.client = null;
  }

  async connect() {
    const transport = new StdioClientTransport({
      command: this.config.serverCommand,
      args: this.config.serverArgs
    });

    this.client = new Client({
      name: "rental-analyzer",
      version: "1.0.0"
    }, {
      capabilities: {}
    });

    await this.client.connect(transport);
  }

  async analyzeProperty(propertyData) {
    const result = await this.client.callTool({
      name: "analyze_property",
      arguments: { propertyData }
    });
    return result;
  }
}

// services/aiService.js - 整合 MCP 的 AI 服務
class AIService {
  constructor(mcpClient) {
    this.mcpClient = mcpClient;
  }

  async analyzeProperty(propertyData) {
    // 透過 MCP 調用 Claude 分析
    return await this.mcpClient.analyzeProperty(propertyData);
  }

  async analyzeImage(imageFile) {
    // 直接調用 Gemini Pro 2.5
    return await this.gemini.analyzeImage(imageFile);
  }

  async searchProperties(criteria) {
    return await this.gemini.batchSearch(criteria);
  }
}
```

#### 本地存儲實現
```javascript
// services/storageService.js
class StorageService {
  constructor() {
    this.dbName = 'RentalAnalyzer';
    this.version = 1;
  }
  
  async init() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, this.version);
      
      request.onupgradeneeded = (event) => {
        const db = event.target.result;
        const store = db.createObjectStore('properties', { 
          keyPath: 'id', 
          autoIncrement: true 
        });
        store.createIndex('score', 'score');
        store.createIndex('createdAt', 'createdAt');
      };
      
      request.onsuccess = (event) => {
        this.db = event.target.result;
        resolve();
      };
    });
  }
  
  async saveProperty(property) {
    const transaction = this.db.transaction(['properties'], 'readwrite');
    const store = transaction.objectStore('properties');
    return store.add({
      ...property,
      createdAt: new Date(),
      favorite: false
    });
  }
  
  async getAllProperties() {
    const transaction = this.db.transaction(['properties'], 'readonly');
    const store = transaction.objectStore('properties');
    return new Promise((resolve) => {
      const request = store.getAll();
      request.onsuccess = () => resolve(request.result);
    });
  }
}
```
